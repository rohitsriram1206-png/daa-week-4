PROGRAM 16: Number of ways to represent n using 1 and 3 (DP)
========================================================================

#include <stdio.h> 
int main() {
   int n;
   scanf("%d", &n);
   long long dp[n + 1];
   for (int i = 0; i <= n; i++)
       dp[i] = 0;
   dp[0] = 1;
   for (int i = 1; i <= n; i++) {
       dp[i] = dp[i - 1];
       if (i >= 3)
           dp[i] += dp[i - 3];
   }
   printf("%lld", dp[n]);
   return 0;
}


========================================================================
PROGRAM 17: Maximum-value path in a chessboard (DP)
========================================================================

#include <stdio.h> 
int max(int a, int b) { 
   return a > b ? a : b; 
} 
int main() {
   int n;
   scanf("%d", &n);
   int board[n][n];
   int dp[n][n];
   for (int i = 0; i < n; i++)
       for (int j = 0; j < n; j++)
           scanf("%d", &board[i][j]);
   dp[0][0] = board[0][0];
   for (int j = 1; j < n; j++)
       dp[0][j] = dp[0][j - 1] + board[0][j];
   for (int i = 1; i < n; i++)
       dp[i][0] = dp[i - 1][0] + board[i][0];
   for (int i = 1; i < n; i++) {
       for (int j = 1; j < n; j++) {
           dp[i][j] = board[i][j] + max(dp[i - 1][j], dp[i][j - 1]);
       }
   }

   printf("%d", dp[n - 1][n - 1]);
   return 0;
}


========================================================================
PROGRAM 18: Longest common subsequence (DP)
========================================================================

#include <stdio.h> 
#include <string.h> 
int max(int a, int b) { 
   return a > b ? a : b; 
} 

int main() {
   char s1[1000], s2[1000];
   scanf("%s", s1);
   scanf("%s", s2);
   int n = strlen(s1);
   int m = strlen(s2);
   int dp[n + 1][m + 1];
   for (int i = 0; i <= n; i++)
       dp[i][0] = 0;
   for (int j = 0; j <= m; j++)
       dp[0][j] = 0;
   for (int i = 1; i <= n; i++) {
       for (int j = 1; j <= m; j++) {
           if (s1[i - 1] == s2[j - 1])
               dp[i][j] = dp[i - 1][j - 1] + 1;
           else
               dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
       }
   }
   printf("%d", dp[n][m]);
   return 0;
}


========================================================================
PROGRAM 19: Longest non-decreasing subsequence (DP)
========================================================================

#include <stdio.h> 
int main() {
   int n;
   scanf("%d", &n);

   int arr[n], dp[n];
   int maxLength = 1;
   for (int i = 0; i < n; i++) {
       scanf("%d", &arr[i]);
       dp[i] = 1;
   }
   for (int i = 1; i < n; i++) {
       for (int j = 0; j < i; j++) {
           if (arr[j] <= arr[i] && dp[i] < dp[j] + 1)
               dp[i] = dp[j] + 1;
       }
       if (dp[i] > maxLength)
           maxLength = dp[i];
   }
   printf("%d", maxLength);
   return 0;
}
